def _stamp(delta=0): return (datetime.now(timezone.utc)+timedelta(seconds=delta)).isoformat()
def _statements(): return [x.strip() for x in SCHEMA_PATH.read_text().split(";\n\n") if x.strip()]
async def _drop():
    async with db_backend.connect(None) as db:
        await db._run_raw("SET FOREIGN_KEY_CHECKS=0")
        for t in TABLES: await db._run_raw(f"DROP TABLE IF EXISTS `{t}`")
        await db._run_raw("SET FOREIGN_KEY_CHECKS=1"); await db.commit()
async def _install():
    async with db_backend.connect(None) as db:
        for s in _statements(): await db._run_raw(s)
        await db.commit()
async def _scalar(sql,params=()):
    async with db_backend.connect(None) as db:
        c=await db._run_raw(sql,params); r=await c.fetchone(); return None if r is None else r[0]
async def _rows(sql,params=()):
    async with db_backend.connect(None) as db:
        c=await db._run_raw(sql,params); return await c.fetchall()
async def _engine(t): return str(await _scalar("SELECT ENGINE FROM information_schema.tables WHERE table_schema=%s AND table_name=%s",(db_backend.mysql_settings()["database"],t)) or "")
async def _indexes(t): return {str(x[0]) for x in await _rows("SELECT DISTINCT INDEX_NAME FROM information_schema.statistics WHERE table_schema=%s AND table_name=%s",(db_backend.mysql_settings()["database"],t))}
async def _fks(t): return {str(x[0]) for x in await _rows("SELECT DISTINCT CONSTRAINT_NAME FROM information_schema.referential_constraints WHERE constraint_schema=%s AND table_name=%s",(db_backend.mysql_settings()["database"],t))}
