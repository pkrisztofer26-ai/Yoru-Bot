async def ensure_community_project(
    self, guild_id: int, *, snapshot: WorldSnapshot | None = None
) -> CommunityProjectSnapshot | None:
    active = await self.current_community_project(guild_id)
    if active is not None:
        return active
    world = snapshot or await self.ensure_current(guild_id)
    definition, city_key = self._project_seed_choice(guild_id, world.cycle_id)
    started = _utcnow()
    deadline = started + timedelta(hours=int(definition.duration_hours))
    async with aiosqlite.connect(self.database.path) as db:
        await db.execute("BEGIN IMMEDIATE")
        # A world cycle may seed at most one shared project. INSERT OR IGNORE
        # makes concurrent/restart creation converge on the same state.
        await db.execute(
            """INSERT OR IGNORE INTO rp_world_community_projects(
                   guild_id,project_key,city_key,source_cycle_id,status,target_units,min_participants,
                   actor_cap_units,contributed_units,participant_count,started_at,deadline_at,completed_at,updated_at
               ) VALUES(?,?,?,?,?,?,?,?,0,0,?,?,NULL,?)""",
            (
                int(guild_id), definition.key, city_key, str(world.cycle_id), "active",
                int(definition.target_units), int(definition.min_participants), int(definition.actor_cap_units),
                _iso(started), _iso(deadline), _iso(started),
            ),
        )
        cur = await db.execute(
            """SELECT project_id,guild_id,project_key,city_key,source_cycle_id,status,target_units,
                      min_participants,actor_cap_units,contributed_units,participant_count,
                      started_at,deadline_at,completed_at
               FROM rp_world_community_projects WHERE guild_id=? AND source_cycle_id=?""",
            (int(guild_id), str(world.cycle_id)),
        )
        row = await cur.fetchone()
        await db.commit()
    if row is None:
        return None
    project = self._community_project_row(row)
    return project if project.status == "active" else None
