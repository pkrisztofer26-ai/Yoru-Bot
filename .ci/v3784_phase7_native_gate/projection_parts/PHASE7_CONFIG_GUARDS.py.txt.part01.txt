@dataclass(frozen=True, slots=True)
class CausalitySignalDefinition:
    key: str
    domain: str
    label: str
    window_hours: int = WORLD_CAUSALITY_DEFAULT_WINDOW_HOURS
    min_participants: int = 3
    min_volume: int = 3
    actor_cap_units: int = 8
    actor_volume_cap: int = 6
    global_cap_units: int = 24
    max_event_units: int = 4
    max_event_volume: int = 3


@dataclass(frozen=True, slots=True)
class CausalityBranchRule:
    story_key: str
    from_beat_key: str
    signal_key: str
    favored_beat_key: str
    opposed_beat_key: str
    max_shift_bp: int = WORLD_CAUSALITY_BRANCH_MAX_SHIFT_BP
